def generate_memory_recall(scenes):
    memories = []

    for scene in scenes:
        for event in scene:
            text = event.get("text", "").strip()
            etype = event.get("type", "")
            importance = event.get("importance", 0)

            # Keep dialogue too, not just high importance
            if text:
                # Optionally, you can prioritize discovery/combat events
                if etype in ["discovery", "combat"] or importance >= 1:
                    memories.append(text)

    # Stitch into one string
    recallable_memory = " ".join(memories)
    return recallable_memory

ef generate_memory_recall(scenes):
    memories = []

    for i, scene in enumerate(scenes):
        print(f"Scene {i}, length {len(scene)}")  # Debug
        for j, event in enumerate(scene):
            text = event.get("text", "").strip()
            etype = event.get("type", "")
            importance = event.get("importance", 0)

            print(f"Event {j}: {etype}, importance={importance}, text='{text[:30]}...'")  # Debug

            if not text:
                continue

            # Keep all events for now
            memories.append(text)

    print("Generated memories count:", len(memories))
    return memories
